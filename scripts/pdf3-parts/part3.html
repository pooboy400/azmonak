core/elo.pyبخش ۱ — ثابت‌ها و تابع انتظار
"""موتور Elo آزمونک — رتبه‌بندی دوطرفه دانش‌آموز و سؤال."""

K_STUDENT_START = 32    # K دانش‌آموز در شروع
K_QUESTION_START = 8    # K سؤال در شروع
D_SCALE = 400.0         # ثابت مقیاس لجستیک؛ هر ۴۰۰ امتیاز = شانس ۱۰ برابر


def expected_success(r_student: float, r_question: float) -> float:
    """شانس درست‌زدن دانش‌آموز روی سؤال — فرمول لجستیک Elo."""
    return 1.0 / (1.0 + 10.0 ** ((r_question - r_student) / D_SCALE))


def k_student(total_answers: int) -> int:
    """K متغیر دانش‌آموز: ۳۲ تا ۵۰ پاسخ، ۲۴ تا ۱۵۰، بعد ۱۶."""
    if total_answers < 50:
        return 32
    if total_answers < 150:
        return 24
    return 16


def k_question(total_answers: int) -> int:
    """K متغیر سؤال: هر پاسخ دربارهٔ سؤال اطلاعات کمتری دارد."""
    return K_QUESTION_START if total_answers < 40 else 4
core/elo.pyبخش ۲ — آپدیت دوطرفه پس از هر پاسخ
def apply_result(
    r_student: float,
    r_question: float,
    correct: bool,
    k_s: int = K_STUDENT_START,
    k_q: int = K_QUESTION_START,
) -> tuple[float, float, float]:
    """دانش‌آموز: R' = R + K_s*(S−E) | سؤال: R' = R + K_q*(E−S)

    Returns:
        (توان جدید دانش‌آموز، سختی جدید سؤال، انتظار E قبل از پاسخ)
    """
    e = expected_success(r_student, r_question)
    s = 1.0 if correct else 0.0
    new_student = r_student + k_s * (s - e)
    new_question = r_question + k_q * (e - s)
    return new_student, new_question, e
چرا تابع، سه‌تایی برمی‌گرداند؟ چون سرویس جلسه به E قبل از پاسخ هم نیاز دارد — برای انتخاب سؤال بعدی (فصل ۳) و ذخیره در params_snapshot (G7). یک محاسبه، سه مصرف.
مثال عددی گام‌به‌گام — همان «تست طلایی» سند اصلی
۱دانش‌آموزی با توان 1200 به سؤالی با سختی 1350 می‌رسد. انتظار: E = 1/(1+10^(150/400)) = 1/(1+2.371) ≈ 0.296 — فقط ۲۹٫۶٪ شانس درست‌زدن.
۲او درست جواب می‌دهد! سورپرایز: S − E = 1 − 0.296 = 0.704 — خبر بزرگ، چون کاری بهتر از انتظار انجام شد.
۳دانش‌آموز: 1200 + 32 × 0.704 = ۱۲۲۲٫۵ — جهشی حدود ۲۲ امتیازی؛ با K متغیر، دفعات بعد این جهش کوچک‌تر می‌شود.
۴سؤال: 1350 + 8 × (0.296 − 1) = 1350 − 5.6 = ۱۳۴۴٫۴ — سؤال کمی آسان‌تر شد؛ این همان «رأی جامعه» بر سختی است. (هر دو عدد با تست خودکار بازتولید می‌شوند: python3 core/tests/test_core.py)
دستگیرهمقدار فعلیاگر عوضش کنی چه می‌شود؟
D_SCALE400بزرگ‌تر ← منحنی ملایم‌تر؛ سورپرایزها کوچک‌تر و امتیازها آرام‌تر حرکت می‌کنند
K_STUDENT_START32بزرگ‌تر ← رتبه تندتر بالا/پایین می‌رود ولی پرنوسان‌تر؛ نوسانِ پس از هر جلسه بدتر می‌شود
K_QUESTION_START8بزرگ‌تر ← سختی سؤال‌ها سریع‌تر اصلاح می‌شود ولی یک بچهٔ خوش‌شانس می‌تواند سختی را بپیچاند
آستانه‌های کاهش K50 / 150کم‌تر کردنشان ← رتبه زودتر «آرام» می‌گیرد؛ زیاد کردنشان ← رتبه دیرتر پایدار می‌شود
چرا Elo و نه IRT / Glicko-2 / TrueSkill؟ IRT (نظریهٔ پاسخ به سؤال) جامع‌تر است و در توفل و PISA کار می‌کند، ولی برای کالیبره‌شدن به صدها پاسخ در هر سؤال نیاز دارد — بانک سؤال MVP ما ۵۰ تا ۱۵۰ سؤال است و این سرمایه وجود ندارد. Glicko-2 و TrueSkill «عدم‌قطعیت» را هم مدل می‌کنند ولی پیچیدگی‌شان برای موتوری که باید یک دانش‌آموز دبیرستانی بتواند توضیحش بدهد، هزینه است نه فایده. Elo با ۳۰ خط پایتون، شفاف و قابل دفاع است و اگر روزی داده انبوه شد، مهاجرت به IRT از روی همان دو ستون توان/سختی ممکن است.

جایابی — کجا شروع کنی؟
۴ سؤال، ۳ رده سختی، یک عدد قابل اعتماد (تصمیم D1)

اولین روزی که وارد آزمونک می‌شوی، سیستم هیچ‌چیز درباره‌ات نمی‌داند. سه راه رایج وجود دارد: خوداظهاری («خودت بگو چه‌قدر بلدی»)، پیش‌فرض ثابت (به همه ۱۲۰۰ بده) و پیش‌آزمون. هر سه رد شده‌اند: خوداظهاری سوگیری دارد، پیش‌فرض ثابت برای نصف بچه‌ها اشتباه است و پیش‌آزمون هزینه و استرس دارد. انتخاب شد جلسهٔ جایابی: خودِ موتور Elo با ۴ سؤال، سطح واقعی‌ات را پیدا کند — همان کاری را بکند که در بقیهٔ عمرش می‌کند، فقط با K بزرگ‌تر و ثابت.

🧒اگر یک بچهٔ ۱۰ ساله بپرسد: «چطور بازی از سطح من سر در می‌آورد؟»

مثل بازی‌ای که اولش از تو می‌پرسد «تا حالا بازی کردی؟» و بعد چند مرحلهٔ آزمایشی می‌دهد: آسانی می‌دهد، اگر راحت رد کردی سخت‌تر می‌دهد، اگر سختش گیر کردی برمی‌گردد عقب. آزمونک هم اول ۴ سؤال از سه پلهٔ مختلف می‌دهد — پلهٔ متوسط، پلهٔ آسان، پلهٔ سخت — و با جواب‌هایت می‌فهمد «قدِ» امتیازیت حدوداً چقدر است.

یک راز کوچک: ترتیب این سؤال‌ها پراکنده است، نه اول آسان و بعد سخت! چون اگر ترتیب منظم باشد، بچه‌ها الگو را می‌فهمند و شروع می‌کنند به حدس‌زدنِ سیستماتیک: «جونِ سؤال قبلی سخت بود، این یکی حتماً آسان است». پراکندگی این تقلب را می‌بندد.

💡مفهوم — جایابی، همان Elo است با پدال گاز

هیچ فرمول جدیدی این‌جا وجود ندارد؛ همان آپدیت دوطرفهٔ فصل ۱ است با سه تفاوت حساب‌شده. تفاوت اول: نقطهٔ شروع ۱۲۵۰ است — وسطِ ناحیهٔ ممکن، بی‌طرف نسبت به هیچ‌کس. تفاوت دوم: سختی ۴ سؤال روی سه ردهٔ ۱۱۵۰ / ۱۲۵۰ / ۱۳۵۰ پخش می‌شود (سه سؤال اول) و سؤال چهارم از رده‌ای انتخاب می‌شود که به برآورد جاری نزدیک است؛ این‌طوری هر پاسخ، برآورد را واقعاً جابه‌جا می‌کند. تفاوت سوم: در طول این ۴ پاسخ، K ثابت روی ۳۲ می‌ماند (تصمیم D1) — می‌خواهیم سریع به سطح واقعی برسیم؛ «آرام‌گیری» K شروع جلسهٔ بعد است. خروجی جایابی دو چیز است: توان اولیه R_s و یک شاخص عدم‌قطعیت که با هر پاسخ کوچک‌تر می‌شود؛ اگر دفعه‌های بعد شاهد پرش‌های عجیب بودیم، همین شاخص به سیستم می‌گوید هنوز مطمئن نیستیم.

👨‍💻مهندسی — فایل core/placement.pyهمان آپدیت Elo، با ترتیب پراکنده و K ثابت
"""جایابی ۴ سؤالی (D1) — پیدا کردن سطح واقعی دانش‌آموز تازه‌وارد."""

from .elo import apply_result

START_RATING = 1250.0    # برآورد خنثی قبل از اولین پاسخ
PLACEMENT_K = 32         # K ثابت در طول جایابی (تصمیم D1)
QUESTION_K = 8           # K سؤال در جایابی
# سه رده سختی با «ترتیب پراکنده» — دنبالهٔ صعودی الگو می‌سازد و حدس می‌آورد
SPREAD = (1250.0, 1150.0, 1350.0)
BASE_UNCERTAINTY = 100.0  # شاخص عدم‌قطعیت پایه = نصف پراکندگی
UNCERTAINTY_DECAY = 15.0  # کاهش عدم‌قطعیت با هر پاسخ
MIN_UNCERTAINTY = 40.0


def next_question_rating(index: int, r_student: float) -> float:
    """سختی سؤال iاُم: سه سؤال اول پخش روی رده‌ها، چهارمی نزدیک برآورد."""
    if index < len(SPREAD):
        return SPREAD[index]
    return min(1350.0, max(1150.0, r_student))


def run_placement(answers: list[bool]) -> dict:
    """کل جلسهٔ جایابی؛ خروجی: R_s اولیه + شاخص عدم‌قطعیت + ردپا."""
    r_student = START_RATING
    trace = []
    for i, correct in enumerate(answers):
        r_question = next_question_rating(i, r_student)
        new_student, new_question, e = apply_result(
            r_student, r_question, correct, PLACEMENT_K, QUESTION_K
        )
        trace.append({
            "step": i + 1, "r_question": r_question,
            "expected": round(e, 3), "correct": correct,
            "r_student": round(r_student, 1),
            "r_student_new": round(new_student, 1),
        })
        r_student = new_student
    uncertainty = max(
        MIN_UNCERTAINTY,
        BASE_UNCERTAINTY - UNCERTAINTY_DECAY * len(answers),
    )
    return {"r_student": round(r_student, 1),
            "uncertainty": uncertainty, "trace": trace}
مثال عددی گام‌به‌گام — جلسهٔ جایابی با الگوی «درست، غلط، درست، درست»
۱سؤال 1250 (ردهٔ میانی): E = 0.500؛ درست → 1250 + 32×0.5 = 1266.
۲سؤال 1150 (پراکنده به عقب): E = 0.661؛ غلط → 1266 − 32×0.661 = 1244.8 — غلطِ «غیرمنتظره» سقوط می‌آورد.
۳سؤال 1350: E = 0.353؛ درست → 1244.8 + 32×0.647 = 1265.5 — جهشِ زیر فشار.
۴سؤال چهارم از نزدیک‌ترین رده به برآورد: 1266؛ E = 0.5؛ درست → 1281.5. خروجی: توان اولیه ۱۲۸۱٫۵ + عدم‌قطعیت ۴۰.
چرا جایابی «کوچک» و نه آزمون تطبیقی کامل (CAT)؟ آزمون تطبیقی استاندارد، هر سؤال را با بهینه‌سازی آماری برمی‌گزیند و به ده‌ها سؤال و بانک کالیبره نیاز دارد. جایابی ۴ سؤالی، ۸۰٪ فایده را با ۵٪ هزینه می‌گیرد و چون از همان موتور Elo استفاده می‌کند، هیچ گذار و ناسازگاری بعد از جلسهٔ اول وجود ندارد؛ کد هم یک حلقهٔ ۴ خطی است.
