"""SM-2 آزمونک — مرور فاصله‌دار با کیفیت سه‌سیگنالی."""

EF_START = 2.5      # آسانی نسبی اولیه
EF_FLOOR = 1.3      # کف EF — هرگز زیر آن نمی‌رود
T_FAST = 15         # ثانیه؛ زیر آن: حدس رفلکسی (سقف q=۴)
T_SLOW = 70         # ثانیه؛ بالای آن: سخت به جواب رسیده (−۰٫۵)
BASE_CORRECT = 4.0  # q پایه برای پاسخ درست
BASE_WRONG = 2.0    # q پایه برای غلط (زیر ۳ ← ریست فاصله)
CONFIDENT_BONUS = 1.0  # «مطمئن بودم» +۱
GUESS_CAP = 4.0     # سقف q برای «حدس می‌زدم ولی درست»


def quality(correct: bool, confident: bool | None,
            elapsed_sec: float) -> float:
    """کیفیت q بین ۰ تا ۵ از سه سیگنال پاسخ."""
    q = BASE_CORRECT if correct else BASE_WRONG
    if confident is True:
        q = min(5.0, q + CONFIDENT_BONUS)
    elif confident is False and correct:
        q = min(q, GUESS_CAP)  # حدسِ درست، حافظه واقعی قوی‌تر نیست
    if elapsed_sec < T_FAST:
        q = min(q, 4.0)        # پاسخ آنی مشکوک به حدس رفلکسی
    elif elapsed_sec > T_SLOW:
        q -= 0.5               # تلاش طولانی = یادگیری متزلزل‌تر
    return max(0.0, min(5.0, q))

def update_ef(ease_factor: float, q: float) -> float:
    """EF' = EF + 0.1 − d·0.08 − d²·0.02 که d = 5 − q؛ کف ۱٫۳."""
    d = 5.0 - q
    new_ef = ease_factor + 0.1 - d * 0.08 - d * d * 0.02
    return max(EF_FLOOR, new_ef)


def next_interval(repetition: int, ease_factor: float, q: float,
                  previous_interval: int = 0) -> int:
    """فاصلهٔ مرور بعدی برحسب روز؛ q<3 یعنی ریست به ۱ روز."""
    if q < 3.0:
        return 1
    if repetition <= 1:
        return 1
    if repetition == 2:
        return 6
    return max(1, round(previous_interval * ease_factor))
جدا‌سازی حالت از منطق: این فایل «بی‌حالت» است — repetition، ease_factor و previous_interval را از دیتابیس می‌گیرد و عدد برمی‌گرداند. به همین دلیل تست‌کردنش بی‌نیاز از سرور است و هسته (core) طبق قاعدهٔ پروژه هیچ I/O ندارد.
مثال عددی گام‌به‌گام — یک آیتم مرور در سه سناریو (خروجی واقعی تست)
۱سناریوی درستِ مطمئن: درست + «مطمئن بودم» + ۳۰ ثانیه ← q = min(5, 4+1) = 5 ← EF: 2.5 → 2.6 (فقط +۰٫۱) و زنجیره: 1 → 6 → 15 → 38.
۲سناریوی حدسِ درستِ سریع: درست + «حدس می‌زدم» + ۱۰ ثانیه ← دو سقف هم‌زمان: q = 4 ← EF: 2.5 → 2.5 — ثابت! سیستم نه جریمه می‌کند نه جایزه؛ حدسِ درست رشد مرور نمی‌خرد.
۳سناریوی غلطِ مطمئن: غلط + «مطمئن بودم» + ۳۰ ثانیه ← q = 2+1 = 3 ← EF: 2.5 → 2.36 و چون q ≥ 3 زنجیره حفظ می‌شود؛ اما همان غلط بدون اعتماد ← q = 2 < 3 ← فاصلهٔ بعدی ۱ روز (ریست کامل) با EF → 2.18.
۴کف EF: آیتمی با EF=۱٫۳۵ که q=۰ بگیرد ← جریمهٔ d=۵ یعنی 1.35 − 0.8 = 0.55، ولی کف اعمال می‌شود: EF = 1.3 — سادگیِ نسبی هرگز زیر کف نمی‌رود تا آیتم «غیرقابل‌بخشش» نشود.

چرا SM-2 و نه Leitner / Pimsleur / FSRS / HLR؟ لایتنر فقط پنج جعبهٔ ثابت دارد — برای «چه زمانی مرور» بسیار خشن است. Pimsleur برای واژگان صوتی طراحی شده و تعمیمش به ریاضی و علوم محدود است. FSRS و HLR (دوئولینگو) مدل‌های رگرسیونی/عصبی مدرنی‌اند که به دادهٔ واقعی هر کاربر برای کالیبراسیون نیاز دارند — در MVP داده نداریم. SM-2 نقطهٔ طلایی است: بدون نیاز به داده شروع می‌شود، از روز اول منطقی رفتار می‌کند و بعداً با دادهٔ جمع‌شده می‌تواند به FSRS ارتقا یابد.

گام اول — نمرهٔ جلسه (D13). هر سؤال، وزنی برابر با ردهٔ سختی‌اش می‌گیرد: آسان (۱۱۵۰) ×۱، متوسط (۱۲۵۰) ×۱٫۵، سخت (۱۳۵۰) ×۲. نمرهٔ جلسه یعنی سهمِ وزنیِ درست‌ها از کل وزن:

Score = Σ ( wi × si ) ÷ Σ wi × 100نمرهٔ وزن‌دار جلسه — وزن تابع سختی Elo سؤال؛ سیگنال اعتماد عمداً این‌جا نیست

گام دوم — رگرسیون روند. پیش‌بینی، میانگینِ وزن‌دارِ نماییِ نمره‌های جلسات است: جلسهٔ آخر وزن λ⁰ = 1، یکی مانده‌به‌آخر λ = 0.85، دوتا مانده λ² ≈ 0.72 و به همین ترتیب. λ = ۰٫۸۵ یعنی «حافظهٔ بلندمدت با احتیاط»: جلسات اخیر حاکم‌اند ولی گذشته هنوز صدا دارد. پیش از ۸ جلسه، پیش‌بینی اصلاً نمایش داده نمی‌شود — عددِ کم‌پایه اعتماد می‌سوزاند. توان R_s هم جداگانه در نمودار روند رسم می‌شود (D15): نمره برای دانش‌آموز، Rs برای روایت دفاع.

"""پیش‌بینی نمره — نمرهٔ وزن‌دار D13 + رگرسیون نمایی λ."""

LAMBDA = 0.85            # وزن نمایی جلسات اخیر
MIN_SESSIONS = 8         # حداقل جلسه برای نمایش پیش‌بینی
TIER_WEIGHTS = {1150: 1.0, 1250: 1.5, 1350: 2.0}


def question_weight(r_question: float) -> float:
    """وزن سؤال از سختی Elo — نزدیک‌ترین ردهٔ ۱۱۵۰/۱۲۵۰/۱۳۵۰."""
    nearest_tier = min(TIER_WEIGHTS, key=lambda t: abs(t - r_question))
    return TIER_WEIGHTS[nearest_tier]


def session_score(answers: list[tuple[bool, float]]) -> float:
    """نمرهٔ وزن‌دار: Σ(وزن×درستی) ÷ Σ(وزن) × ۱۰۰."""
    total_weight = 0.0
    earned_weight = 0.0
    for correct, r_question in answers:
        w = question_weight(r_question)
        total_weight += w
        if correct:
            earned_weight += w
    if total_weight == 0:
        return 0.0
    return earned_weight / total_weight * 100.0


def predict_next_score(session_scores: list[float],
                       lam: float = LAMBDA) -> dict:
    """میانگین نمایی وزن‌دار؛ i=۰ یعنی آخرین جلسه."""
    if len(session_scores) < MIN_SESSIONS:
        return {"available": False,
                "reason": f"حداقل {MIN_SESSIONS} جلسه لازم است"}
    weights = [lam ** i for i in range(len(session_scores))]
    forecast = (sum(w * s for w, s in zip(weights, session_scores))
                / sum(weights))
    naive_mean = sum(session_scores) / len(session_scores)
    return {"available": True, "forecast": round(forecast, 1),
            "naive_mean": round(naive_mean, 1),
            "direction": "up" if forecast >= naive_mean else "down"}
مثال عددی گام‌به‌گام — همان مثال G1 سند (خروجی واقعی تست)
۱جلسه‌ای با وزن‌های (۶ آسان ×۱، ۳ متوسط ×۱٫۵، ۱ سخت ×۲) ← کل وزن Σw = 12.5.
۲هفت درستِ «۴ آسان + ۲ متوسط + ۱ سخت» ← (4×1 + 2×1.5 + 1×2) ÷ 12.5 × 100 = 9 ÷ 12.5 × 100 = ۷۲.
۳همان هفت درست اما اگر همگی از سؤال‌های آسان بود ← 7 × 1 ÷ 12.5 × 100 = ۵۶ — سختی واقعاً دیده می‌شود؛ دقیقاً چیزی که نمرهٔ خام نمی‌فهمد.
۴پیش‌بینی با نمره‌های آزمایشی [52, 58, 60, 65, 63, 70, 68, 74] ← میانگین نمایی 61.4 در برابر میانگین سادهٔ 63.8؛ چون جلسات اخیر گاهی پایین‌ترند، پیش‌بینی حساب‌شده محافظه‌کارتر از میانگین بی‌وزن است.
چرا میانگین نمایی و نه رگرسیون خطی / LSTM؟ رگرسیون خطی روی نمره‌های ناپایدارِ ۸ جلسه‌ایِ اول، به نوسان‌ها بیش‌ازحد واکنش نشان می‌دهد و LSTM به هزاران داده نیاز دارد. میانگین نمایی با یک پارامتر (λ) رفتار «حافظهٔ اخیرگرا» را دقیق کنترل می‌کند، در آزمایش علمی قابل تفسیر است و اگر بد پیش‌بینی کرد، معلوم است کدام پارامتر را بچرخانی.
