"""نردبان سطح دسته‌ای (D16) — سه لیگ برای هر دستهٔ سؤال."""

DROP_STREAK = 2            # n: غلط پیاپی لازم برای سقوط
RISE_STREAK = 2            # m: درست پیاپی لازم برای ارتقا
MIN_LEVEL, MAX_LEVEL = 1, 3
LEVEL_BANDS = {1: 1150, 2: 1250, 3: 1350}  # نوار سختی هر سطح
STUDY_ALERT_SUCCESS = 0.30  # آستانهٔ بنر مطالعه
ALERT_WINDOW = 5            # پنجرهٔ محاسبهٔ موفقیت اخیر
START_LEVEL = 2             # شروع هر دسته از میانه (پیش‌فرض مهندسی)


class CategoryLadder:
    """وضعیت نردبان یک دسته برای یک دانش‌آموز."""

    def __init__(self, category: str) -> None:
        self.category = category
        self.level = START_LEVEL
        self.wrong_streak = 0      # شمارندهٔ غلط پیاپی
        self.right_streak = 0      # شمارندهٔ درست پیاپی
        self.weakness_counter = 0  # در کف سطح ۱: شمارش افت بی‌اثر
        self.promoted_this_session = False
        self.recent: list[bool] = []   # پنجرهٔ ۵ پاسخ آخر
        self.alert = False

    def difficulty_band(self) -> float:
        """نوار سختی Elo این سطح — ورودی فیلتر انتخاب سؤال."""
        return LEVEL_BANDS[self.level]

    def record(self, correct: bool) -> dict:
        """ثبت یک پاسخ و به‌روزرسانی سطح؛ خروجی: گزارش رویداد."""
        event = {"moved": False, "direction": None, "alert": False}
        self.recent.append(correct)
        self.recent = self.recent[-ALERT_WINDOW:]

        if correct:
            self.right_streak += 1
            self.wrong_streak = 0
            if (self.right_streak >= RISE_STREAK
                    and self.level < MAX_LEVEL
                    and not self.promoted_this_session):
                self.level += 1
                self.right_streak = 0
                self.promoted_this_session = True
                event.update(moved=True, direction="up")
        else:
            self.wrong_streak += 1
            self.right_streak = 0
            if self.wrong_streak >= DROP_STREAK:
                self.wrong_streak = 0
                if self.level > MIN_LEVEL:
                    self.level -= 1
                    self.promoted_this_session = False
                    event.update(moved=True, direction="down")
                else:
                    self.weakness_counter += 1

        if self.level == MIN_LEVEL and len(self.recent) == ALERT_WINDOW:
            rate = sum(self.recent) / len(self.recent)
            self.alert = rate < STUDY_ALERT_SUCCESS
            event["alert"] = self.alert
        return event

SM-2 — مرور فاصله‌دار
قبل از آن‌که فراموش کنی مرورت کن: ۱ ← ۶ ← ۱۵ روز و بعد هر بار طولانی‌تر

Elo و نردبان دربارهٔ «فهم» تو حرف می‌زنند، ولی حافظه قانون خودش را دارد: هرچه یاد گرفته‌ای، اگر تمرین نشود، طبق یک منحنی ثابت محو می‌شود. SM-2 — الگوریتمی که پشت اپ‌هایی مثل Anki نشسته — این منحنی را با دو عدد مهار می‌کند: آسانی نسبی EF (این آیتم برای تو چقدر «چسبنده» است) و شمارهٔ تکرار n. بعد از هر پاسخ، فقط همین دو عدد به‌روز می‌شوند و فاصلهٔ مرور بعدی از دلشان بیرون می‌آید.

🧒اگر یک بچهٔ ۱۰ ساله بپرسد: «چرا بعضی سؤال‌ها برمی‌گردند سراغم؟»

مثل آبیاری گل است! اگر هر روز به گل آب بدهی، ریشه‌اش تنبل می‌شود و سطحی می‌ماند. اگر سه هفته فراموشش کنی، می‌خشکد. گل‌خورد هر روز آب نمی‌خواهد؛ دقیقاً لحظه‌ای که دارد تشنه می‌شود آب می‌خواهد. سؤال‌ها هم همین‌طورند: روز اول که یاد گرفتی، فردا دوباره می‌آیند؛ اگر خوب جوابشان بدهی، دفعهٔ بعد شش روز بعد می‌آیند؛ باز هم خوب جواب بدهی، پانزده روز بعد؛ و هر بار فاصله‌ها طولانی‌تر می‌شود چون مغزت این‌ها را «محکم‌تر» نگه داشته.

حالا نکتهٔ باحالش: گل‌ها با هم فرق دارند! بعضی سؤال‌ها «کاکتوس»‌اند — سخت‌گیر می‌شوی با آن‌ها و آب کم می‌خواهند؛ بعضی «بنفشه»‌اند — مدام تشنه‌اند. آن عدد EF همین است: برای هر سؤال، جداگانه می‌گوید چقدر ماندگار بوده. و اگر یک روز به یک سؤال کامل سربازنزدی؟ فاصله‌اش ریست می‌شود به فردا — مثل اینکه گل پژمیده و باید از اول بهش برسی.

💡مفهوم — کیفیت سه‌سیگنالی و فرمول درجه‌دو

کیفیت هر پاسخ، q نام دارد و بین ۰ تا ۵ است؛ در آزمونک از سه سیگنال ساخته می‌شود، نه فقط درست/غلط: ① درستی — پایه: درست ← ۴، غلط ← ۲. ② اعتماد (D14) — «مطمئن بودم» یک امتیاز اضافه می‌کند (سقف ۵)؛ «حدس می‌زدم» ولی درست، سقف q را در ۴ می‌بندد — چون حدسِ درست نباید حافظهٔ واقعی قوی‌تری نشان بدهد. ③ تأخیر (D10) — پاسخِ زیر ۱۵ ثانیه مشکوک به حدس رفلکسی است (سقف q = ۴) و بالای ۷۰ ثانیه یعنی سخت به جواب رسیده‌ای (−۰٫۵). سیگنال اعتماد فقط همین‌جا و در قرینهٔ بدفهمی مصرف می‌شود و وارد نمرهٔ جلسه نمی‌شود — قاعدهٔ ثابت سند.

دو عدد هر آیتم با q به‌روز می‌شوند. اول آسانی نسبی؛ اگر d = 5 − q را باز کنی، فرمول SM-2 شکل حساب‌شده‌اش را نشان می‌دهد:

EF′ = EF + ( 0.1 − d · 0.08 − d2 · 0.02 )     d = 5 − q   ,   EF ≥ 1.3قانون آپدیت آسانی نسبی — جریمه درجه‌دو برای کیفیت بد؛ کف ۱٫۳

سه خاصیت از همین بازکردن بیرون می‌آید: در q = ۴، د = ۱ است پس 0.1 − 0.08 − 0.02 = 0 و EF دست‌نخورده می‌ماند؛ در q = ۵ فقط ۰٫۱ اضافه می‌شود (رشد آهسته و صبور)؛ و هرچه q بدتر شود، جملهٔ d² سنگین‌تر از جریمهٔ خطی رشد می‌کند — کیفیتِ خیلی بد، سقوط EF را تندتر می‌کند. سپس فواصل:

q < 3 ⇒ reset  |   I(1) = 1  ,   I(2) = 6  ,   I(n) = I(n−1) × EFزنجیرهٔ مرور — یک غلطِ واقعی (q زیر ۳) زنجیره را به روز اول برمی‌گرداند
شکل ۴ — زنجیرهٔ مرور یک آیتم سالم؛ هر پرش = ضرب در EF همان آیتم

یک ظرافت اجرایی: آستانه‌های تأخیر (۱۵ و ۷۰ ثانیه) پیش‌فرض سراسری‌اند و هر سؤال می‌تواند با فیلد est_time_sec خودش آن‌ها را override کند — سؤال محاسباتی سنگین حق دارد زمان بیشتری بخواهد. و «مطمئن اما غلط» (q پایینِ با اعتماد بالا) مهم‌ترین سیگنال انفرادی کل محصول است: همان قرینهٔ بدفهمی که در گزارش جلسه ثبت می‌شود.
