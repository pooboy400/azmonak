    19→// ---------- deterministic PRNG ----------
    20→function mulberry32(seed: number) {
    21→  let a = seed >>> 0;
    22→  return function () {
    23→    a |= 0;
    24→    a = (a + 0x6d2b79f5) | 0;
    25→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    26→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    27→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    28→  };
    29→}
    30→const rng = mulberry32(1405);
    31→const rand = (min: number, max: number) => min + rng() * (max - min);
    32→const randInt = (min: number, max: number) => Math.floor(rand(min, max + 1));
    33→const pick = <T,>(arr: T[]): T => arr[Math.floor(rng() * arr.length)];
    34→const clamp = (v: number, lo: number, hi: number) => Math.min(hi, Math.max(lo, v));
    35→const round1 = (v: number) => Math.round(v * 10) / 10;

    92→// ---------- content selection ----------
    93→function pickTopic(topics: TopicState[]): TopicState {
    94→  // weaker topics get priority (plan 1.2 — educational value weight)
    95→  const weights = topics.map((t) => 5 - t.level + rng() * 2);
    96→  const total = weights.reduce((a, b) => a + b, 0);
    97→  let r = rng() * total;
    98→  for (let i = 0; i < topics.length; i++) {
    99→    r -= weights[i];
   100→    if (r <= 0) return topics[i];
   101→  }
   102→  return topics[topics.length - 1];
   103→}

   105→function shuffled<T>(arr: T[]): T[] {
   106→  const a = [...arr];
   107→  for (let i = a.length - 1; i > 0; i--) {
   108→    const j = Math.floor(rng() * (i + 1));
   109→    [a[i], a[j]] = [a[j], a[i]];
   110→  }
   111→  return a;
   112→}

   114→/**
   115→ * Question selection with preference order: preferred topic → other topics,
   116→ * each at its own level. 1st pass respects exposure window + in-session
   117→ * uniqueness; fallback (small sample bank) only respects in-session
   118→ * uniqueness. Returns the topic actually used so the D16 ladder updates it.
   119→ */
   120→function selectQuestion(
   121→  topicStates: TopicState[],
   122→  preferred: TopicState | null,
   123→  catalog: TopicRow[],
   124→  seen: string[],
   125→  sessionPicks: Set<string>
   126→): { qid: string; diff: Diff; topic: TopicState } | null {
   127→  const rest = preferred ? topicStates.filter((t) => t !== preferred) : topicStates;
   128→  const ordered = preferred ? [preferred, ...shuffled(rest)] : shuffled(topicStates);
   129→  for (const ts of ordered) {
   130→    const row = catalog.find((r) => r.topicId === ts.topicId)!;
   131→    const want = DIFF_BY_LEVEL[ts.level - 1];
   132→    for (const diff of [want, ...DIFF_BY_LEVEL.filter((d) => d !== want)]) {
   133→      const pool = (row.questions.get(diff) ?? []).filter((qid) => !seen.includes(qid) && !sessionPicks.has(qid));
   134→      if (pool.length > 0) return { qid: pick(pool), diff, topic: ts };
   135→    }
   136→  }
   137→  for (const ts of ordered) {
   138→    const row = catalog.find((r) => r.topicId === ts.topicId)!;
   139→    for (const diff of DIFF_BY_LEVEL) {
   140→      const pool = (row.questions.get(diff) ?? []).filter((qid) => !sessionPicks.has(qid));
   141→      if (pool.length > 0) return { qid: pick(pool), diff, topic: ts };
   142→    }
   143→  }
   144→  return null;
   145→}

   147→// ---------- one attempt ----------
   148→interface WalkCtx {
   149→  userId: string;
   150→  subjectRating: number;
   151→  answered: number; // per student-subject
   152→  seen: string[]; // exposure window (G8) — capped at windowW below
   153→  sessionPicks: Set<string>; // in-session uniqueness
   154→  windowW: number; // effective exposure window (sample banks are smaller than the real 150-question bank)
   155→  promoted: Set<string>; // max one promotion per session per topic (D16)
   156→}
   157→
   158→function simulateAttempt(
   159→  ctx: WalkCtx,
   160→  qid: string,
   161→  diff: Diff,
   162→  skill: number,
   163→  topic: TopicState | null,
   164→  time: number,
   165→  sessionId: string,
   166→  order: number
   167→): { correct: boolean; weight: number; latency: number; endTime: number } {
   168→  const q = qState.get(qid)!;
   169→  const pCorrect = clamp(0.62 + 2.0 * (skill - TIER_SKILL[diff]), 0.1, 0.9);
   170→  const correct = rng() < pCorrect;
   171→
   172→  // D14: confidence self-report
   173→  const confidence = correct
   174→    ? rng() < 0.78
   175→      ? "CONFIDENT"
   176→      : "GUESSED"
   177→    : rng() < 0.35
   178→      ? "CONFIDENT"
   179→      : "GUESSED";
   180→
   181→  // latency shaped by correctness/confidence (t_fast=15s, t_slow=70s — D10)
   182→  let latency: number;
   183→  if (correct && confidence === "CONFIDENT") latency = rand(8, 28);
   184→  else if (correct) latency = rand(18, 50);
   185→  else if (confidence === "CONFIDENT") latency = rand(25, 60);
   186→  else latency = rand(30, 85);
   187→  latency = round1(latency);
   188→
   189→  const selected = correct
   190→    ? q.correctLetter
   191→    : pick(LETTERS.filter((l) => l !== q.correctLetter));
   192→  const weight = WEIGHT[diff];
   193→
   194→  // Elo (two-sided, plan K schedule)
   195→  const rBefore = ctx.subjectRating;
   196→  const E = expected(rBefore, q.rating);
   197→  const ks = kStudent(ctx.answered);
   198→  const kq = kQuestion(q.count);
   199→  const rAfter = correct ? rBefore + ks * (1 - E) : rBefore - ks * E;
   200→  ctx.subjectRating = rAfter;
   201→  q.rating = correct ? q.rating + kq * (E - 1) : q.rating + kq * (1 - E);
   202→  q.count += 1;
   203→  ctx.answered += 1;
   204→  ctx.seen.push(qid);
   205→  ctx.sessionPicks.add(qid);
   206→  if (ctx.seen.length > ctx.windowW) ctx.seen.shift();
   207→
   208→  // D16 level ladder
   209→  if (topic) {
   210→    topic.answered += 1;
   211→    if (correct) {
   212→      topic.rightStreak += 1;
   213→      topic.wrongStreak = 0;
   214→      if (topic.rightStreak >= 2 && !ctx.promoted.has(topic.topicId)) {
   215→        topic.level = Math.min(3, topic.level + 1);
   216→        topic.rightStreak = 0;
   217→        ctx.promoted.add(topic.topicId);
   218→      }
   219→    } else {
   220→      topic.wrongStreak += 1;
   221→      topic.rightStreak = 0;
   222→      if (topic.wrongStreak >= 2) {
   223→        if (topic.level > 1) topic.level -= 1;
   224→        else topic.weakness += 1; // at floor — weakness counter (study-alert input)
   225→        topic.wrongStreak = 0;
   226→      }
   227→    }
   228→  }
   229→
   230→  attemptRows.push({
   231→    id: nextAttemptId(),
   232→    sessionId,
   233→    userId: ctx.userId,
   234→    questionId: q.id,
   235→    orderInSession: order,
   236→    selected,
   237→    isCorrect: correct,
   238→    confidence,
   239→    latencySec: latency,
   240→    weight,
   241→    rBefore: Math.round(rBefore * 10) / 10,
   242→    rAfter: Math.round(rAfter * 10) / 10,
   243→    paramsSnapshot: JSON.stringify({ Ks: ks, Kq: kq, zone: "0.65-0.75", exposure: 20, ver: "plan-v1.3" }),
   244→    isSample: true,
   245→    createdAt: new Date(time),
   246→  });
   247→
   248→  // SM-2 quality (correct, confidence, latency) + state update
   249→  if (topic) {
   250→    const quality = correct
   251→      ? confidence === "CONFIDENT"
   252→        ? latency <= 15
   253→          ? 5
   254→          : 4
   255→        : 3
   256→      : confidence === "CONFIDENT"
   257→        ? 2
   258→        : 1;
   259→    const rv = topic.review;
   260→    if (quality >= 3) {
   261→      rv.rep += 1;
   262→      rv.interval = rv.rep === 1 ? 1 : rv.rep === 2 ? 6 : Math.ceil(rv.interval * rv.ef);
   263→    } else {
   264→      rv.rep = 0;
   265→      rv.interval = 1;
   266→    }
   267→    rv.ef = Math.max(1.3, rv.ef + 0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02));
   268→    rv.last = time;
   269→  }
   270→
   271→  return { correct, weight, latency, endTime: time + (latency + rand(2, 7)) * 1000 };
   272→}

   511→          const rStart = ctx.subjectRating;
   512→          let scoreW = 0;
   513→          let scoreSum = 0;
   514→          let correctCount = 0;
   515→          let attemptsMade = 0;
   516→          let t = slot.time;
   517→
   518→          for (let k = 1; k <= stopAt; k++) {
   519→            let qid: string | null = null;
   520→            let diff: Diff = "MEDIUM";
   521→            let topic: TopicState | null = null;
   522→            if (slot.placement) {
   523→              // spread difficulty across tiers (D1: پراکنده ۱۱۵۰–۱۳۵۰), topics in random order
   524→              const want = DIFF_BY_LEVEL[(k - 1) % 3];
   525→              const shuffled = [...topicStates].sort(() => rng() - 0.5);
   526→              outer: for (const ts of shuffled) {
   527→                const row = catalog.find((r) => r.topicId === ts.topicId)!;
   528→                for (const d of [want, ...DIFF_BY_LEVEL]) {
   529→                  const pool = (row.questions.get(d) ?? []).filter(
   530→                    (cand) => !ctx.seen.includes(cand) && !ctx.sessionPicks.has(cand)
   531→                  );
   532→                  if (pool.length > 0) {
   533→                    qid = pick(pool);
   534→                    diff = d;
   535→                    topic = ts;
   536→                    break outer;
   537→                  }
   538→                }
   539→              }
   540→              if (!qid) break;
   541→            } else {
   542→              const preferred = pickTopic(topicStates);
   543→              const chosen = selectQuestion(topicStates, preferred, catalog, ctx.seen, ctx.sessionPicks);
   544→              if (!chosen) break;
   545→              qid = chosen.qid;
   546→              diff = chosen.diff;
   547→              topic = chosen.topic;
   548→            }
   549→            const res = simulateAttempt(ctx, qid, diff, u.skill, topic, t, sessionId, k);
   550→            attemptsMade += 1;
   551→            if (res.correct) correctCount += 1;
   552→            scoreW += res.weight;
   553→            scoreSum += res.weight * (res.correct ? 1 : 0);
   554→            t = res.endTime;
   555→            if (t > NOW) break; // never leak into the future
   556→          }
   557→          if (attemptsMade === 0) continue; // no orphan sessions
   558→          const completed = !slot.abandoned && attemptsMade === count && t <= NOW;
   559→          sessionRows.push({
   560→            id: sessionId,
   561→            userId: u.id,
   562→            subjectId,
   563→            type: slot.placement ? "PLACEMENT" : "PRACTICE",
   564→            status: completed ? "COMPLETED" : "ABANDONED",
   565→            questionCount: count,
   566→            correctCount,
   567→            score: completed && scoreW > 0 ? round1((scoreSum / scoreW) * 100) : null,
   568→            rStart: Math.round(rStart),
   569→            rEnd: completed ? Math.round(ctx.subjectRating) : null,
   570→            startedAt: new Date(slot.time),
   571→            finishedAt: completed ? new Date(Math.min(t, NOW)) : null,
   572→            lastActivityAt: new Date(Math.min(t, NOW)),
   573→            isSample: true,
   574→            createdAt: new Date(slot.time),
   575→          });
   576→        }
   577→
   578→        // --- final ability + review rows ---
   579→        subjectAbilityRows.push({
   580→          userId: u.id,
   581→          subjectId,
   582→          rating: Math.round(ctx.subjectRating),
   583→          answeredCount: ctx.answered,
   584→          isSample: true,
   585→        });
   586→        for (const ts of topicStates) {
   587→          if (ts.answered === 0) continue;
   588→          topicAbilityRows.push({
   589→            userId: u.id,
   590→            subjectId,
   591→            topicId: ts.topicId,
   592→            level: ts.level,
   593→            wrongStreak: ts.wrongStreak,
   594→            rightStreak: ts.rightStreak,
   595→            weaknessCount: ts.weakness,
   596→            answeredCount: ts.answered,
   597→            isSample: true,
   598→          });
   599→          const last = ts.review.last;
   600→          const due = last !== null ? last + Math.max(1, ts.review.interval) * DAY : NOW - randInt(1, 3) * DAY;
   601→          reviewRows.push({
   602→            userId: u.id,
   603→            subjectId,
   604→            topicId: ts.topicId,
   605→            ef: round1(ts.review.ef * 100) / 100,
   606→            intervalDays: Math.max(1, ts.review.interval),
   607→            repetitions: ts.review.rep,
   608→            dueAt: new Date(Math.min(due, NOW + randInt(0, 2) * DAY)),
   609→            lastReviewedAt: last ? new Date(last) : null,
   610→            isSample: true,
   611→          });
   612→        }
   613→      }
   614→    }
   615→
   616→    // ===== 5. two in-progress sessions (resume + live UI demo) =====
   617→    console.log("⏳ creating in-progress sessions…");
   618→    const inProgress: Array<{ userId: string; code: string; minutesAgo: number; attempts: number }> = [
   619→      { userId: "smpu01", code: "PHY11T", minutesAgo: 180, attempts: 5 },
   620→      { userId: "smpu05", code: "BIO11T", minutesAgo: 90, attempts: 4 },
   621→    ];
   622→    for (const ip of inProgress) {
   623→      const subjectId = subjectIds.get(ip.code)!;
   624→      const u = USERS.find((x) => x.id === ip.userId)!;
   625→      const catalog = catalogs.get(ip.code)!;
   626→      const sessionId = nextSessionId();
   627→      const topicStates: TopicState[] = catalog.map((r) => ({
   628→        topicId: r.topicId,
   629→        level: 2,
   630→        wrongStreak: 0,
   631→        rightStreak: 0,
   632→        weakness: 0,
   633→        answered: 0,
   634→        review: { ef: 2.5, interval: 0, rep: 0, last: null },
   635→      }));
   636→      const ctx: WalkCtx = {
   637→        userId: u.id,
   638→        subjectRating: 1100,
   639→        answered: 0,
   640→        seen: [],
   641→        sessionPicks: new Set(),
   642→        windowW: Math.min(20, Math.max(6, catalog.reduce((n, r) => n + [...r.questions.values()].flat().length, 0) - 4)),
   643→        promoted: new Set(),
   644→      };
   645→      const start = NOW - ip.minutesAgo * MIN;
   646→      let t = start;
   647→      let correctCount = 0;
   648→      for (let k = 1; k <= ip.attempts; k++) {
   649→        const ts = pickTopic(topicStates);
   650→        const chosen = selectQuestion(topicStates, ts, catalog, ctx.seen, ctx.sessionPicks);
   651→        if (!chosen) break;
   652→        const res = simulateAttempt(ctx, chosen.qid, chosen.diff, u.skill, chosen.topic, t, sessionId, k);
   653→        if (res.correct) correctCount += 1;
   654→        t = res.endTime;
   655→      }
   656→      sessionRows.push({
   657→        id: sessionId,
   658→        userId: u.id,
   659→        subjectId,
   660→        type: "PRACTICE",
   661→        status: "ACTIVE",
   662→        questionCount: 10,
   663→        correctCount,
   664→        score: null,
   665→        rStart: 1100,
   666→        rEnd: null,
   667→        startedAt: new Date(start),
   668→        finishedAt: null,
   669→        lastActivityAt: new Date(Math.min(t, NOW)),
   670→        isSample: true,
   671→        createdAt: new Date(start),
   672→      });
   673→    }
   674→
   675→    // ===== 6. batch insert + question drift updates =====
   676→    console.log(`💾 inserting ${sessionRows.length} sessions, ${attemptRows.length} attempts…`);
   677→    const chunk = <T,>(arr: T[], size: number): T[][] => {
   678→      const out: T[][] = [];
   679→      for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
   680→      return out;
   681→    };
   682→    for (const part of chunk(sessionRows, 500)) await db.session.createMany({ data: part });
   683→    for (const part of chunk(attemptRows, 1000)) await db.attempt.createMany({ data: part });
   684→    for (const part of chunk(subjectAbilityRows, 500)) await db.subjectAbility.createMany({ data: part });
   685→    for (const part of chunk(topicAbilityRows, 500)) await db.topicAbility.createMany({ data: part });
   686→    for (const part of chunk(reviewRows, 500)) await db.reviewSchedule.createMany({ data: part });
   687→
   688→    for (const st of qState.values()) {
   689→      await db.question.update({
   690→        where: { id: st.id },
   691→        data: { rating: Math.round(st.rating * 10) / 10, ratingCount: st.count },
   692→      });
   693→    }
   694→
   695→    // ===== 7. summary =====
   696→    const [users, identity, subjects, topics, questions, sessions, attempts, sa, ta, reviews, classes, friendships, mis] =
   697→      await Promise.all([
   698→        db.user.count(),
   699→        db.identity.count(),
   700→        db.subject.count(),
   701→        db.topic.count(),
   702→        db.question.count(),
   703→        db.session.count(),
   704→        db.attempt.count(),
   705→        db.subjectAbility.count(),
   706→        db.topicAbility.count(),
   707→        db.reviewSchedule.count(),
   708→        db.classRoom.count(),
   709→        db.friendship.count(),
   710→        db.misconception.count(),
   711→      ]);
   712→    console.log("\n✅ seed complete — sample data summary:");
   713→    console.table({
   714→      users, identity, subjects, topics, questions, sessions, attempts,
   715→      subjectAbility: sa, topicAbility: ta, reviewSchedule: reviews,
   716→      classes, friendships, misconceptions: mis,
   717→    });
   718→  } finally {
   719→    await db.$disconnect();
   720→  }
   721→}
   722→
   723→main().catch((e) => {
   724→  console.error("❌ seed failed:", e);
   725→  process.exit(1);
   726→});
